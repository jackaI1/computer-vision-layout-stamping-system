   109                                           def _integral_image(mask: np.ndarray) -> np.ndarray:
   110                                               # uint32 is sufficient (the running count maxes out at H*W ink pixels, far
   111                                               # below 2**32) and halves the memory of this per-page table versus int64. The
   112                                               # summed-area identity a-b-c+e stays correct under unsigned wraparound because
   113                                               # every window sum lies in [0, H*W].
   114                                               out = np.zeros((mask.shape[0] + 1, mask.shape[1] + 1), dtype=np.uint32)
   115                                               # Row-first traversal reads the boolean source contiguously. Both passes
   116                                               # write into the final table, avoiding two H*W uint32 temporary arrays.
   117                                               body = out[1:, 1:]
   118                                               np.cumsum(mask, axis=1, dtype=np.uint32, out=body)
   119                                               np.cumsum(body, axis=0, dtype=np.uint32, out=body)
   120                                               return out
